CHECK_CXX_COMPILER_FLAG("-std=c++0x" COMPILER_SUPPORTS_CXX0X)
if(COMPILER_SUPPORTS_CXX11)
   set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++11")
   add_definitions(-DCOMPILEDWITHC11)
   message(STATUS "Using flag -std=c++11.")
elseif(COMPILER_SUPPORTS_CXX0X)
   set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++0x")
   add_definitions(-DCOMPILEDWITHC0X)
   message(STATUS "Using flag -std=c++0x.")
else()
   message(FATAL_ERROR "The compiler ${CMAKE_CXX_COMPILER} has no C++11 support. Please use a different C++ compiler.")
endif()

option(WITH_ONNX_RUNTIME "Build synchronous CPU semantic region masking" OFF)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime installation with include/ and lib/")
if(WITH_ONNX_RUNTIME)
    find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h
        HINTS ${ONNXRUNTIME_ROOT}/include PATH_SUFFIXES onnxruntime/core/session)
    find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime HINTS ${ONNXRUNTIME_ROOT}/lib ${ONNXRUNTIME_ROOT}/lib64)
    if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
        message(FATAL_ERROR "ONNX Runtime not found. Set ONNXRUNTIME_ROOT to its installation prefix.")
    endif()
    # ORT 1.17.x C++ headers require C++14; keep the rest of ORB-SLAM3 on C++11.
    set_source_files_properties(src/SemanticMasker.cc PROPERTIES COMPILE_FLAGS "-std=c++14")
    target_compile_definitions(${PROJECT_NAME} PRIVATE ORB_SLAM3_WITH_ONNX)
    target_include_directories(${PROJECT_NAME} PRIVATE ${ONNXRUNTIME_INCLUDE_DIR})
    target_link_libraries(${PROJECT_NAME} ${ONNXRUNTIME_LIBRARY})
    get_filename_component(ONNXRUNTIME_LIB_DIR ${ONNXRUNTIME_LIBRARY} DIRECTORY)
    set_property(TARGET ${PROJECT_NAME} APPEND PROPERTY BUILD_RPATH ${ONNXRUNTIME_LIB_DIR})
endif()
